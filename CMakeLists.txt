cmake_minimum_required(VERSION 3.25)

option(HDR_BUILD_GUI   "Build the ImGui desktop application" ON)
option(HDR_BUILD_CLI   "Build the command line tool"         ON)
option(HDR_BUILD_TESTS "Build unit tests"                    ON)
option(HDR_WARNINGS_AS_ERRORS "Treat compiler warnings as errors (used by CI)" OFF)
option(HDR_ENABLE_ASAN "Build with AddressSanitizer (MSVC /fsanitize=address)" OFF)

# Only install the vcpkg dependencies of the parts being built (must precede project()).
if(HDR_BUILD_GUI)
  list(APPEND VCPKG_MANIFEST_FEATURES "gui")
endif()
if(HDR_BUILD_CLI)
  list(APPEND VCPKG_MANIFEST_FEATURES "cli")
endif()
if(HDR_BUILD_TESTS)
  list(APPEND VCPKG_MANIFEST_FEATURES "tests")
endif()

project(HDR_DNG_Merger VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

# Put all executables + DLLs next to each other so vcpkg app-local deps work.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# Link-time optimisation for optimised builds where the toolchain supports it.
include(CheckIPOSupported)
check_ipo_supported(RESULT HDR_IPO_SUPPORTED OUTPUT HDR_IPO_ERROR LANGUAGES CXX)
if(HDR_IPO_SUPPORTED)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO ON)
endif()

if(MSVC)
  add_compile_options(/W4 /permissive- /MP /utf-8 /Zc:__cplusplus /EHsc)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  if(HDR_WARNINGS_AS_ERRORS)
    add_compile_options(/WX)
  endif()
  if(HDR_ENABLE_ASAN)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
  endif()
else()
  add_compile_options(-Wall -Wextra)
  if(HDR_WARNINGS_AS_ERRORS)
    add_compile_options(-Werror)
  endif()
  if(HDR_ENABLE_ASAN)
    add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address)
  endif()
endif()

add_compile_definitions(HDR_VERSION_STRING="${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}")

# Windows executables get the shared application manifest (UTF-8 code page, long paths)
# and a version resource.
function(hdr_configure_windows_app target description)
  if(WIN32)
    set(HDR_RC_NAME ${target})
    set(HDR_RC_DESCRIPTION "${description}")
    set(rc_file ${CMAKE_CURRENT_BINARY_DIR}/${target}_version.rc)
    configure_file(${PROJECT_SOURCE_DIR}/cmake/version.rc.in ${rc_file} @ONLY)
    target_sources(${target} PRIVATE ${PROJECT_SOURCE_DIR}/cmake/app.manifest ${rc_file})
  endif()
endfunction()

add_subdirectory(src/core)
if(HDR_BUILD_CLI)
  add_subdirectory(src/cli)
endif()
if(HDR_BUILD_GUI)
  add_subdirectory(src/gui)
endif()
if(HDR_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
